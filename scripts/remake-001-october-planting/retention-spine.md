## PHA D — Verify cơ học (chạy TRƯỚC audit nặng)

```python
import re
tts = re.search(r"## TTS SCRIPT\s*(.*)", open(PATH, encoding="utf-8").read(), re.DOTALL).group(1)
W = lambda s: len(re.sub(r"[#*\-`>\[\]()_]", " ", s).split())
total = W(tts); sec = lambda i: W(tts[:i]) / 2.5

# 1. Do dai
print("words", total, "| phut", round(total/150, 1), "->", "OK" if 3000 <= total <= 3500 else "FAIL")

# 2. L6 — bang chung < giay 20  (so viet CHU nen bat chu, khong bat digit)
proof = re.search(r"\b(fifty|four inches|one to two weeks|Minnesota|South Dakota)\b", tts, re.I)
print("L6 proof @", round(sec(proof.start()), 1), "s ->", "OK" if proof and sec(proof.start()) < 20 else "FAIL")

# 3. L4 — loop-cue song giay 40-55 (khong duoc la intro)
seg = tts[int(40*2.5*5):int(56*2.5*6)]
print("L4 self-ID cue ->", "OK" if re.search(r"before November|after\b.*November", seg, re.I) else "FAIL")

# 4. DELIVERY-LOCK — con so 16 va 7 deu phai xuat hien
print("16 ->", "OK" if re.search(r"\bsixteen\b", tts, re.I) else "FAIL",
      "| 7 ->", "OK" if re.search(r"\bseven\b", tts, re.I) else "FAIL")

# 5. Micro-bridge phu du 15 ranh gioi
items = len(re.findall(r"\bNumber (one|two|three|four|five|six|seven|eight|nine|ten|eleven|twelve|thirteen|fourteen|fifteen|sixteen)\b", tts, re.I))
print("items", items, "-> ", "OK" if items == 16 else "FAIL (phai dung 16)")

# 6. X5 — narrator integrity
bad = re.findall(r"\bI (grew|planted|tested|tried)\b|\bin my garden\b", tts, re.I)
print("X5 ->", "OK" if not bad else "FAIL " + str(bad))
```
